Problem 1

I used simulation of groups of people to estimate the probability that at least two people share a birthday. Assuming that birthdays are independent and equal likely.

has_shared_birthday = function(group_size) {
  birthdays = sample(1:365, size = group_size, replace = TRUE)
  any(duplicated(birthdays))
}
set.seed(8105)

birthday_probabilities =
  expand_grid(
    group_size = 2:50,
    simulation = 1:10000
  ) |>
  mutate(shared_birthday = map_lgl(group_size, has_shared_birthday)) |>
  group_by(group_size) |>
  summarize(
    probability = mean(shared_birthday),
    .groups = "drop"
  )
birthday_probabilities |>
  ggplot(aes(x = group_size, y = probability)) +
  geom_line() +
  geom_point(size = 1.5) +
  geom_hline(yintercept = 0.5, linetype = "dashed", color = "gray50") +
  scale_y_continuous(labels = scales::label_percent()) +
  labs(
    title = "Probability of a shared birthday",
    x = "Number of people",
    y = "Estimated probability"
  ) +
  theme_minimal(base_size = 12)

The estimated probability of a shared birthday increases with group size. It is approximately 50% for a group of 23 people.

Problem 3:

simulate_t_test = function(true_mean) {
  sample_data = rnorm(n = 30, mean = true_mean, sd = 5)

  t.test(sample_data, mu = 0, alternative = "two.sided") |>
    broom::tidy() |>
    select(estimate, p.value)
}
set.seed(8105)

t_test_results =
  expand_grid(
    true_mean = 0:6,
    simulation = 1:5000
  ) |>
  mutate(test_result = map(true_mean, simulate_t_test)) |>
  unnest(test_result)

t_test_summary =
  t_test_results |>
  group_by(true_mean) |>
  summarize(
    rejection_rate = mean(p.value < 0.05),
    mean_all = mean(estimate),
    mean_rejected = mean(estimate[p.value < 0.05]),
    .groups = "drop"
  )
t_test_summary |>
  ggplot(aes(x = true_mean, y = rejection_rate)) +
  geom_line() +
  geom_point() +
  scale_x_continuous(breaks = 0:6) +
  scale_y_continuous(labels = scales::label_percent()) +
  labs(
    title = "Rejection rate of the one-sample t-test",
    x = "True mean",
    y = "Proportion of tests rejecting the null"
  ) +
  theme_minimal(base_size = 12)

When the true mean was 0, about 5% of the tests rejected the null hypothesis. This is close to the expected Type I error rate. As the true mean increased, the test was more likely to reject the null, and power was almost 100% when the true mean was 4.

t_test_summary |>
  pivot_longer(
    c(mean_all, mean_rejected),
    names_to = "sample_set",
    values_to = "average_estimate"
  ) |>
  mutate(
    sample_set = if_else(
      sample_set == "mean_all",
      "All samples",
      "Significant results only"
    )
  ) |>
  ggplot(aes(x = true_mean, y = average_estimate, color = sample_set)) +
  geom_abline(slope = 1, intercept = 0, linetype = "dashed", color = "gray50") +
  geom_line() +
  geom_point() +
  scale_x_continuous(breaks = 0:6) +
  labs(
    title = "Average estimates with and without selection",
    x = "True mean",
    y = "Average estimated mean",
    color = NULL
  ) +
  theme_minimal(base_size = 12) +
  theme(legend.position = "bottom")

When I used all the samples, the average estimate was close to the true mean. When I kept only significant results, the average estimate was higher for small positive true means. For example, when the true mean was 1, the average among significant results was about 2.3. By the time the true mean reached 4, almost all tests were significant, so the two averages were close.
